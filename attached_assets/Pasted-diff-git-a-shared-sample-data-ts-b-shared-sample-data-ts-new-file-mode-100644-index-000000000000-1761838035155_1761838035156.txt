+export function searchSampleCompanies(query: string, limit = 10): SampleCompanyData[] {
+  const trimmed = query.trim().toLowerCase();
+  if (!trimmed) {
+    return [];
+  }
+
+  return SAMPLE_COMPANIES.filter((company) => company.name.toLowerCase().includes(trimmed)).slice(0, limit);
+}
+
+export function searchSampleNcms(query: string, limit = 10): SampleNcmSummary[] {
+  const trimmed = query.trim().toLowerCase();
+  if (!trimmed) {
+    return [];
+  }
+
+  return SAMPLE_NCMS.filter((item) =>
+    item.code.toLowerCase().includes(trimmed) || item.description.toLowerCase().includes(trimmed),
+  ).slice(0, limit);
+}
+
+export function getSampleCompanyById(id: number): SampleCompanyData | undefined {
+  return SAMPLE_COMPANIES.find((company) => company.id === id);
+}
+
+export function getSampleNcmByCode(code: string): SampleNcmSummary | undefined {
+  const normalized = code.trim().toLowerCase();
+  return SAMPLE_NCMS.find((item) => item.code.toLowerCase() === normalized);
+}
